#!/bin/bash --login

#SBATCH -J submit_common_uncommon
#SBATCH -p himem
#SBATCH -t 7-0
#SBATCH -n 1
#SBATCH --mem=64G

set -euo pipefail

module load apps/binapps/anaconda3/2024.10
conda activate venv

# ============================================================
# Central settings: edit the paths/experiment as required
# ============================================================
PROJECT_ROOT="/mnt/hum01-home01/d34195bc/projects/n-gram-av-methods"
SCRIPT_ROOT="${PROJECT_ROOT}/scripts"
JOB_SCRIPT="${PROJECT_ROOT}/jobscripts/common_and_uncommon/run_common_and_uncommon_scoring.txt"

DATA_ROOT="/mnt/hum01-home01/d34195bc/av_datasets_updated"
MODEL_ROOT="/mnt/hum01-home01/d34195bc/models"
EXPERIMENT_ROOT="/mnt/hum01-home01/d34195bc/av_datasets_updated_experiments"
EXPERIMENT_NAME="common_and_uncommon"

SAVE_SUBDIR="raw"
COMPLETED_SUBDIR="raw"
ERROR_SUBDIR="errors"
PROBLEM_LIST_NAME="problem_doc_list.txt"

# These two files are expected in ${SCRIPT_ROOT}
MULTI_SCRIPT="score_ngrams_multi.py"
SCORE_SCRIPT_NAME="score_ngrams_distributions"

# Scoring options. For optional --max_len, leave MAX_LEN empty.
MIN_LEN="2"
MAX_LEN=""
LOWERCASE=true
REMOVE_PUNCTUATION=true
REMOVE_SYMBOLS=true
REMOVE_NUMERIC=true
REMOVE_URL=true

# Separate logs from the previous concat-logprobs experiments
LOG_ROOT="/mnt/hum01-home01/d34195bc/slurm_logs/common_and_uncommon"
mkdir -p "${LOG_ROOT}"

CHAIN_JOBS=false
CHILD_PARTITION="himem"
CHILD_TIME="7-0"
CHILD_NTASKS="1"
CHILD_MEM="64G"

# ============================================================
# Corpus, data type, model and context selections
# ============================================================
MODEL_LIST=("gpt2")

# Small run: ONLY Perverted Justice and Enron Cleaned at 10 context tokens
CORPORA=(
    "Perverted Justice"
    "Enron Cleaned"
)

DATA_TYPES=("test" "training")

FIXED_NUM_TOKENS=(10)

# ============================================================
# Helpers
# ============================================================
short_corpus() {
    case "$1" in
        "ACL")               echo "acl" ;;
        "All-the-news")      echo "news" ;;
        "Amazon")            echo "amzn" ;;
        "Enron Cleaned")     echo "enron" ;;
        "IMDB")              echo "imdb" ;;
        "Koppel's Blogs")    echo "blogs" ;;
        "Perverted Justice") echo "pj" ;;
        "StackExchange")     echo "stack" ;;
        "The Apricity")      echo "apric" ;;
        "TripAdvisor")       echo "trip" ;;
        "Wiki")              echo "wiki" ;;
        "Yelp")              echo "yelp" ;;
        *)                   echo "$1" | tr '[:upper:] ' '[:lower:]_' ;;
    esac
}

short_data_type() {
    case "$1" in
        "training") echo "tr" ;;
        "test")     echo "te" ;;
        *)          echo "$1" ;;
    esac
}

count_nonempty_lines() {
    awk 'NF {c++} END {print c+0}' "$1"
}

# ============================================================
# Submit one array per corpus / data type / model
# One array task = one problem, with contexts from FIXED_NUM_TOKENS
# ============================================================
PREV_JOB_ID=""

for MODEL in "${MODEL_LIST[@]}"; do
    for DATA_TYPE in "${DATA_TYPES[@]}"; do
        for CORPUS in "${CORPORA[@]}"; do
            PROBLEM_LIST="${DATA_ROOT}/${DATA_TYPE}/${CORPUS}/${PROBLEM_LIST_NAME}"
            if [[ ! -f "${PROBLEM_LIST}" ]]; then
                echo "Skipping missing problem list: ${PROBLEM_LIST}"
                continue
            fi

            NUM_TASKS="$(count_nonempty_lines "${PROBLEM_LIST}")"
            if [[ "${NUM_TASKS}" -eq 0 ]]; then
                echo "Skipping empty problem list: ${PROBLEM_LIST}"
                continue
            fi

            JOB_NAME="cu_$(short_data_type "${DATA_TYPE}")_$(short_corpus "${CORPUS}")_${MODEL}"
            JOB_LOG_DIR="${LOG_ROOT}/${DATA_TYPE}/${CORPUS}/${MODEL}"
            mkdir -p "${JOB_LOG_DIR}"

            echo "------------------------------------------------------------"
            echo "Corpus: ${CORPUS} | Type: ${DATA_TYPE} | Model: ${MODEL}"
            echo "Tasks: ${NUM_TASKS} | Job: ${JOB_NAME} | Contexts: ${FIXED_NUM_TOKENS[*]}"
            echo "Logs: ${JOB_LOG_DIR}"

            SBATCH_ARGS=(
                --parsable
                --job-name "${JOB_NAME}"
                --array "1-${NUM_TASKS}"
                -p "${CHILD_PARTITION}"
                -t "${CHILD_TIME}"
                -n "${CHILD_NTASKS}"
                --mem "${CHILD_MEM}"
                -o "${JOB_LOG_DIR}/%x_%A_%a.out"
                -e "${JOB_LOG_DIR}/%x_%A_%a.err"
            )

            if [[ "${CHAIN_JOBS}" == "true" && -n "${PREV_JOB_ID}" ]]; then
                SBATCH_ARGS+=(--dependency="afterok:${PREV_JOB_ID}")
            fi

            JOB_ID="$(
                CORPUS="${CORPUS}" \
                DATA_TYPE="${DATA_TYPE}" \
                MODEL="${MODEL}" \
                PROJECT_ROOT="${PROJECT_ROOT}" \
                SCRIPT_ROOT="${SCRIPT_ROOT}" \
                DATA_ROOT="${DATA_ROOT}" \
                MODEL_ROOT="${MODEL_ROOT}" \
                EXPERIMENT_ROOT="${EXPERIMENT_ROOT}" \
                EXPERIMENT_NAME="${EXPERIMENT_NAME}" \
                SAVE_SUBDIR="${SAVE_SUBDIR}" \
                COMPLETED_SUBDIR="${COMPLETED_SUBDIR}" \
                ERROR_SUBDIR="${ERROR_SUBDIR}" \
                PROBLEM_LIST_NAME="${PROBLEM_LIST_NAME}" \
                MULTI_SCRIPT="${MULTI_SCRIPT}" \
                SCORE_SCRIPT_NAME="${SCORE_SCRIPT_NAME}" \
                FIXED_NUM_TOKENS="${FIXED_NUM_TOKENS[*]}" \
                MIN_LEN="${MIN_LEN}" \
                MAX_LEN="${MAX_LEN}" \
                LOWERCASE="${LOWERCASE}" \
                REMOVE_PUNCTUATION="${REMOVE_PUNCTUATION}" \
                REMOVE_SYMBOLS="${REMOVE_SYMBOLS}" \
                REMOVE_NUMERIC="${REMOVE_NUMERIC}" \
                REMOVE_URL="${REMOVE_URL}" \
                sbatch "${SBATCH_ARGS[@]}" "${JOB_SCRIPT}"
            )"
            echo "Submitted job: ${JOB_ID}"

            if [[ "${CHAIN_JOBS}" == "true" ]]; then
                PREV_JOB_ID="${JOB_ID}"
            fi
        done
    done
done

echo "All submissions complete."
