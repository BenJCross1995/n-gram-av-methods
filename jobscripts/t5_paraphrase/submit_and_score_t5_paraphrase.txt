#!/bin/bash --login

#SBATCH -J submit_t5_paraphrase
#SBATCH -p himem
#SBATCH -t 7-0
#SBATCH -n 1
#SBATCH --mem=64G

set -euo pipefail

# Load conda module
module load apps/binapps/anaconda3/2024.10

# Activate the environment
conda activate venv

# ============================================================
# Central shared settings
# ============================================================

PROJECT_ROOT="/mnt/hum01-home01/d34195bc/projects/n-gram-av-methods"
SCRIPT_ROOT="${PROJECT_ROOT}/scripts"

DATA_ROOT="/mnt/hum01-home01/d34195bc/av_datasets"
MODEL_ROOT="/mnt/hum01-home01/d34195bc/models"
EXPERIMENT_ROOT="/mnt/hum01-home01/d34195bc/av_datasets_experiments"

# Generic array jobscript that actually does the work
JOB_SCRIPT="${PROJECT_ROOT}/jobscripts/t5_paraphrase/run_and_score_t5_paraphrase.txt"

# Experiment naming
EXPERIMENT_NAME="t5_paraphrase"

RAW_SUBDIR="raw"
SCORED_SUBDIR="scored"

RAW_ERROR_SUBDIR="errors"
SCORED_ERROR_SUBDIR="scored_errors"

# Problem list file name to look for
PROBLEM_LIST_NAME="agg_problem_list.txt"

# Python scripts
PARAPHRASE_SCRIPT="t5_paraphrase.py"
SCORE_SCRIPT="t5_paraphrase_scoring.py"

# ============================================================
# Which stages to run
# ============================================================

RUN_PARAPHRASE=true
RUN_SCORE=true

# ============================================================
# Paraphrase settings
# ============================================================

MIN_LEN=3
MAX_LEN=""

GREATEST_COMMON=true

LOWERCASE=true
LOWERCASE_INPUT=true
LOWERCASE_OUTPUT=true

MAX_LEFT_EXPANSION=2
MAX_RIGHT_EXPANSION=2
CONTEXT_TOKENS=128

NUM_BEAMS=40
NUM_BEAM_GROUPS=10
NUM_RETURN_SEQUENCES=20
DIVERSITY_PENALTY=0.5
MAX_NEW_TOKENS=20

# Optional smoke-test limits.
# Leave blank to process the full problem.
MAX_NGRAMS=""
MAX_OCCURRENCES_PER_NGRAM=""

# ============================================================
# Metric-scoring settings
# ============================================================

# These are folder names under MODEL_ROOT.
# Change these if your local model folders use different names.
SEMANTIC_MODEL="all-mpnet-base-v2"
NLI_MODEL="nli-deberta-v3-base"
FLUENCY_MODEL="gpt2"

SEMANTIC_WINDOWS=(
    0 5 10
)

NLI_WINDOWS=(
    0 5 10
)

FLUENCY_CONTEXT_TOKENS=10
PHRASE_CONTEXT_TOKENS=10

METRICS_BATCH_SIZE=32

# Keep local original/candidate text windows in the scored RDS.
KEEP_LOCAL_TEXT=false

# Optional smoke-test limit for metric scoring.
# Leave blank to score every generated row.
MAX_SCORE_ROWS=""

# ============================================================
# Logs
# ============================================================

# Files saved as:
# ${LOG_ROOT}/${EXPERIMENT_NAME}/${DATA_TYPE}/${CORPUS}/${MODEL}/${T5_MODEL}/
# <job_name>_<array_job_id>_<array_task_id>.{out,err}

LOG_ROOT="/mnt/hum01-home01/d34195bc/slurm_logs"
mkdir -p "${LOG_ROOT}"

# Whether to chain each submitted job after the last one
CHAIN_JOBS=false

# Shared SLURM settings for child jobs
CHILD_PARTITION="himem"
CHILD_TIME="7-0"
CHILD_NTASKS="1"
CHILD_MEM="64G"

# ============================================================
# Lists to edit
# ============================================================

# Model/tokenizer used to define the common token n-grams
MODEL_LIST=(
    "gpt2"
)

# T5 model used to generate paraphrase candidates
T5_MODEL_LIST=(
    "t5-small"
)

# Removed Reddit and The Telegraph since not in paper
CORPORA=(
    "Wiki"
)

DATA_TYPES=(
    "test"
)

# ============================================================
# Helpers
# ============================================================

short_corpus() {
    case "$1" in
        "ACL")               echo "acl" ;;
        "All-the-news")      echo "news" ;;
        "Amazon")            echo "amzn" ;;
        "Enron Cleaned")     echo "enron" ;;
        "IMDB")              echo "imdb" ;;
        "Koppel's Blogs")    echo "blogs" ;;
        "Perverted Justice") echo "pj" ;;
        "Reddit")            echo "reddit" ;;
        "StackExchange")     echo "stack" ;;
        "The Apricity")      echo "apric" ;;
        "The Telegraph")     echo "tele" ;;
        "TripAdvisor")       echo "trip" ;;
        "Wiki")              echo "wiki" ;;
        "Yelp")              echo "yelp" ;;
        *)                   echo "$1" | tr '[:upper:] ' '[:lower:]_' ;;
    esac
}

short_data_type() {
    case "$1" in
        "training") echo "tr" ;;
        "test")     echo "te" ;;
        *)          echo "$1" ;;
    esac
}

short_t5_model() {
    case "$1" in
        "t5-small") echo "t5s" ;;
        "t5-base")  echo "t5b" ;;
        "t5-large") echo "t5l" ;;
        *)          echo "$1" | tr '[:upper:] /' '[:lower:]__' ;;
    esac
}

count_nonempty_lines() {
    local file="$1"
    awk 'NF {c++} END {print c+0}' "$file"
}

# Turn arrays into space-separated strings for environment passing.
SEMANTIC_WINDOWS_STRING="${SEMANTIC_WINDOWS[*]}"
NLI_WINDOWS_STRING="${NLI_WINDOWS[*]}"

# ============================================================
# Submit loop
# ============================================================

PREV_JOB_ID=""

for MODEL in "${MODEL_LIST[@]}"; do
    for T5_MODEL in "${T5_MODEL_LIST[@]}"; do
        for DATA_TYPE in "${DATA_TYPES[@]}"; do
            for CORPUS in "${CORPORA[@]}"; do

                PROBLEM_LIST="${DATA_ROOT}/${DATA_TYPE}/${CORPUS}/${PROBLEM_LIST_NAME}"

                if [[ ! -f "${PROBLEM_LIST}" ]]; then
                    echo "Skipping missing problem list:"
                    echo "  ${PROBLEM_LIST}"
                    echo
                    continue
                fi

                NUM_TASKS="$(count_nonempty_lines "${PROBLEM_LIST}")"

                if [[ "${NUM_TASKS}" -eq 0 ]]; then
                    echo "Skipping empty problem list:"
                    echo "  ${PROBLEM_LIST}"
                    echo
                    continue
                fi

                JOB_NAME="$(short_data_type "${DATA_TYPE}")_$(short_corpus "${CORPUS}")_${MODEL}_$(short_t5_model "${T5_MODEL}")"

                JOB_LOG_DIR="${LOG_ROOT}/${EXPERIMENT_NAME}/${DATA_TYPE}/${CORPUS}/${MODEL}/${T5_MODEL}"
                mkdir -p "${JOB_LOG_DIR}"

                echo "------------------------------------------------------------"
                echo "Submitting:"
                echo "  EXPERIMENT            = ${EXPERIMENT_NAME}"
                echo "  CORPUS                = ${CORPUS}"
                echo "  DATA_TYPE             = ${DATA_TYPE}"
                echo "  MODEL                 = ${MODEL}"
                echo "  T5_MODEL              = ${T5_MODEL}"
                echo "  PROBLEM_LIST          = ${PROBLEM_LIST}"
                echo "  NUM_TASKS             = ${NUM_TASKS}"
                echo "  JOB_NAME              = ${JOB_NAME}"
                echo "  LOG_DIR               = ${JOB_LOG_DIR}"
                echo "  RUN_PARAPHRASE        = ${RUN_PARAPHRASE}"
                echo "  RUN_SCORE             = ${RUN_SCORE}"
                echo "  RAW_SUBDIR            = ${RAW_SUBDIR}"
                echo "  SCORED_SUBDIR         = ${SCORED_SUBDIR}"
                echo "  SEMANTIC_MODEL        = ${SEMANTIC_MODEL}"
                echo "  NLI_MODEL             = ${NLI_MODEL}"
                echo "  FLUENCY_MODEL         = ${FLUENCY_MODEL}"
                echo "  SEMANTIC_WINDOWS      = ${SEMANTIC_WINDOWS_STRING}"
                echo "  NLI_WINDOWS           = ${NLI_WINDOWS_STRING}"

                SBATCH_ARGS=(
                    --parsable
                    --job-name "${JOB_NAME}"
                    --array "1-${NUM_TASKS}"
                    -p "${CHILD_PARTITION}"
                    -t "${CHILD_TIME}"
                    -n "${CHILD_NTASKS}"
                    --mem "${CHILD_MEM}"
                    -o "${JOB_LOG_DIR}/%x_%A_%a.out"
                    -e "${JOB_LOG_DIR}/%x_%A_%a.err"
                )

                if [[ "${CHAIN_JOBS}" == "true" && -n "${PREV_JOB_ID}" ]]; then
                    SBATCH_ARGS+=(--dependency="afterok:${PREV_JOB_ID}")
                    echo "  DEPENDS ON           = ${PREV_JOB_ID}"
                fi

                JOB_ID="$(
                    CORPUS="${CORPUS}" \
                    DATA_TYPE="${DATA_TYPE}" \
                    MODEL="${MODEL}" \
                    T5_MODEL="${T5_MODEL}" \
                    PROJECT_ROOT="${PROJECT_ROOT}" \
                    SCRIPT_ROOT="${SCRIPT_ROOT}" \
                    DATA_ROOT="${DATA_ROOT}" \
                    MODEL_ROOT="${MODEL_ROOT}" \
                    EXPERIMENT_ROOT="${EXPERIMENT_ROOT}" \
                    EXPERIMENT_NAME="${EXPERIMENT_NAME}" \
                    RAW_SUBDIR="${RAW_SUBDIR}" \
                    SCORED_SUBDIR="${SCORED_SUBDIR}" \
                    RAW_ERROR_SUBDIR="${RAW_ERROR_SUBDIR}" \
                    SCORED_ERROR_SUBDIR="${SCORED_ERROR_SUBDIR}" \
                    PROBLEM_LIST_NAME="${PROBLEM_LIST_NAME}" \
                    PARAPHRASE_SCRIPT="${PARAPHRASE_SCRIPT}" \
                    SCORE_SCRIPT="${SCORE_SCRIPT}" \
                    RUN_PARAPHRASE="${RUN_PARAPHRASE}" \
                    RUN_SCORE="${RUN_SCORE}" \
                    MIN_LEN="${MIN_LEN}" \
                    MAX_LEN="${MAX_LEN}" \
                    GREATEST_COMMON="${GREATEST_COMMON}" \
                    LOWERCASE="${LOWERCASE}" \
                    LOWERCASE_INPUT="${LOWERCASE_INPUT}" \
                    LOWERCASE_OUTPUT="${LOWERCASE_OUTPUT}" \
                    MAX_LEFT_EXPANSION="${MAX_LEFT_EXPANSION}" \
                    MAX_RIGHT_EXPANSION="${MAX_RIGHT_EXPANSION}" \
                    CONTEXT_TOKENS="${CONTEXT_TOKENS}" \
                    NUM_BEAMS="${NUM_BEAMS}" \
                    NUM_BEAM_GROUPS="${NUM_BEAM_GROUPS}" \
                    NUM_RETURN_SEQUENCES="${NUM_RETURN_SEQUENCES}" \
                    DIVERSITY_PENALTY="${DIVERSITY_PENALTY}" \
                    MAX_NEW_TOKENS="${MAX_NEW_TOKENS}" \
                    MAX_NGRAMS="${MAX_NGRAMS}" \
                    MAX_OCCURRENCES_PER_NGRAM="${MAX_OCCURRENCES_PER_NGRAM}" \
                    SEMANTIC_MODEL="${SEMANTIC_MODEL}" \
                    NLI_MODEL="${NLI_MODEL}" \
                    FLUENCY_MODEL="${FLUENCY_MODEL}" \
                    SEMANTIC_WINDOWS="${SEMANTIC_WINDOWS_STRING}" \
                    NLI_WINDOWS="${NLI_WINDOWS_STRING}" \
                    FLUENCY_CONTEXT_TOKENS="${FLUENCY_CONTEXT_TOKENS}" \
                    PHRASE_CONTEXT_TOKENS="${PHRASE_CONTEXT_TOKENS}" \
                    METRICS_BATCH_SIZE="${METRICS_BATCH_SIZE}" \
                    KEEP_LOCAL_TEXT="${KEEP_LOCAL_TEXT}" \
                    MAX_SCORE_ROWS="${MAX_SCORE_ROWS}" \
                    sbatch "${SBATCH_ARGS[@]}" "${JOB_SCRIPT}"
                )"

                echo "  SUBMITTED ID          = ${JOB_ID}"
                echo

                if [[ "${CHAIN_JOBS}" == "true" ]]; then
                    PREV_JOB_ID="${JOB_ID}"
                fi

            done
        done
    done
done

echo "All submissions complete."
